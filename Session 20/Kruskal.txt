#include<bits/stdc++.h>
using namespace std;

class DisjointSet {
    vector<int> parent, rank;

public:
    DisjointSet(int n) {
        parent.resize(n);
        rank.resize(n, 0);
        for(int i = 0; i < n; i++)
            parent[i] = i;
    }

    int findUPar(int node) {
        if(node == parent[node])
            return node;
        return parent[node] = findUPar(parent[node]);
    }

    void unionByRank(int u, int v) {
        int pu = findUPar(u);
        int pv = findUPar(v);

        if(pu == pv) return;

        if(rank[pu] < rank[pv])
            parent[pu] = pv;
        else if(rank[pv] < rank[pu])
            parent[pv] = pu;
        else {
            parent[pv] = pu;
            rank[pu]++;
        }
    }
};

int main() {

    int V, E;
    cin >> V >> E;

    vector<tuple<int,int,int>> edges;

    for(int i = 0; i < E; i++) {
        int u, v, w;
        cin >> u >> v >> w;
        edges.push_back({w, u, v});
    }

    sort(edges.begin(), edges.end());

    DisjointSet ds(V);

    int minCost = 0;

    for(auto it : edges) {
        int w = get<0>(it);
        int u = get<1>(it);
        int v = get<2>(it);

        if(ds.findUPar(u) != ds.findUPar(v)) {
            minCost += w;
            ds.unionByRank(u, v);
        }
    }

    cout << minCost << endl;
}
